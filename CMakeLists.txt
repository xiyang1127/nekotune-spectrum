cmake_minimum_required(VERSION 3.28)
project(NekoTuneSpectrum VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core)
find_package(Threads REQUIRED)
find_package(NekoTuneSDK 2.2 QUIET)
set(NEKOTUNE_SDK_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nekotune-sdk" CACHE PATH "NekoTune SDK 2.2 source directory")
if(NOT NekoTuneSDK_FOUND)
    if(NOT EXISTS "${NEKOTUNE_SDK_SOURCE_DIR}/include/nekotune/sdk/pcm_provider.h")
        message(FATAL_ERROR "Requires SDK 2.2. Set NEKOTUNE_SDK_SOURCE_DIR or CMAKE_PREFIX_PATH.")
    endif()
    add_subdirectory("${NEKOTUNE_SDK_SOURCE_DIR}" nekotune-sdk)
endif()
add_library(spectrum_analysis STATIC src/analyzer.cpp)
target_include_directories(spectrum_analysis PUBLIC src)
target_link_libraries(spectrum_analysis PUBLIC NekoTune::SDK)
nekotune_add_extension(nekotune_spectrum MANIFEST spectrum/extension.json
    SOURCES src/plugin.cpp FILES spectrum/qml/Settings.qml)
target_link_libraries(nekotune_spectrum PRIVATE spectrum_analysis Threads::Threads)
add_custom_target(extension_package ALL DEPENDS nekotune_spectrum_package)
string(REGEX MATCH "^[0-9]+" NEKOTUNE_SDK_COMPILER_MAJOR "${CMAKE_CXX_COMPILER_VERSION}")
set(NEKOTUNE_SDK_HOST_VERSION "")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/build-info.json.in"
    "${CMAKE_BINARY_DIR}/packages/nekotune.spectrum-build-info.json" @ONLY)

if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(analysis_test tests/analysis_test.cpp)
    target_link_libraries(analysis_test PRIVATE spectrum_analysis Qt6::Test)
    add_test(NAME analysis COMMAND analysis_test)
    add_executable(spectrum_plugin_test tests/plugin_test.cpp)
    target_link_libraries(spectrum_plugin_test PRIVATE NekoTune::SDK Qt6::Test)
    add_dependencies(spectrum_plugin_test nekotune_spectrum)
    add_test(NAME plugin_lifecycle COMMAND spectrum_plugin_test $<TARGET_FILE:nekotune_spectrum>)
endif()
